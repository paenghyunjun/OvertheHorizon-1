씬 연결 방법

  1. GameScene에 빈 오브젝트를 만들고 RoundManager를 붙입니다. 이것만으로 1스테이지가 돌아갑니다.
  2. 적 프리팹이 생기면 Rounds → Wave Groups / Boss / Summon Groups의 Prefab 칸에 넣습니다. 비워 두면 RoundManager의 Frames 칸에 넣은 기본 스프라이트 적이 나옵니다.
  3. PullingManager, EnemyBulletPool, ScoreManager는 씬에 없어도 처음 쓰일 때 자동으로 만들어집니다. 미리 채워 둘 개수를 정하고 싶으면 씬에 놓고 Preloads에 프리팹과 개수를 적습니다.
  4. 점수 UI는 씬에 배치한 텍스트에서 ScoreManager.Instance.OnScoreChanged를 구독하면 됩니다. 라운드 표시는 RoundManager.Instance의 OnRoundStarted, OnWaveStarted, OnBossSpawned, OnRoundCleared, OnGameCleared를 씁니다.
  5. 라운드를 추가하려면 Rounds 배열 크기를 늘립니다. 마지막 라운드의 보스가 최종 보스입니다.

임의로 정한 값

기획서에 수치가 없거나 서로 달라서 임시로 정한 부분입니다.

  • 처치 점수: 기획서에 점수 수치가 없습니다. 보스 흡수 점수 비율(찌르레기 1 : 벌매 2)을 따라 100 / 200, 보스는 10000으로 두었습니다.
  • 소환몹 점수 1.5배: 기획서의 "소환된 몹을 잡으면 필살기 게이지 1.5배"를 점수에도 똑같이 적용했습니다.
  • 소환 수: 기획서 앞부분은 찌르레기 8 + 벌매 3, 페이즈 규칙 쪽은 찌르레기 10 + 벌매 2입니다. 더 자세한 뒤쪽 값을 썼습니다.
  • 플레이어 탄 피해 4: 캐릭터 공격력(3~5)이 PlayerData에 아직 없어 Enemy의 Damage Per Hit에 4를 넣었습니다.
  • 미구현: 검독수리 스킬 5종과 페이즈, 흡수 강화, 벌매 분열탄, 필살기 게이지 보상은 이번 범위가 아니라 들어 있지 않습니다. 보스는 기존 패턴 4종을 돌립니다.

Pulling Manager.cs

Assets/Scripts/Manager/Pulling Manager.cs

프리팹 종류와 상관없이 쓰는 공용 오브젝트 풀. 탄막 게임은 탄이 초당 수십 개씩 생기고 사라지므로 Instantiate/Destroy 대신 꺼내 쓰고 되돌려 놓는다. 플레이어 탄, 적 탄(EnemyBulletPool 경유), 적(RoundManager 경유)이 전부 이 매니저 하나를 거친다.

using System.Collections.Generic;
using UnityEngine;

public class PullingManager : MonoBehaviour
{
    // Inspector 에서 "이 프리팹을 몇 개 미리 만들어 둘지" 적는 한 줄.
    [System.Serializable]
    public class Preload
    {
        public GameObject prefab;
        public int count = 20;
    }

    // 프리팹 하나당 하나씩 생기는 보관함.
    private class Bucket
    {
        // 이 프리팹으로 만든 오브젝트를 모아 두는 부모(하이어라키 정리용).
        public Transform root;
        // 쉬고 있는(비활성) 오브젝트.
        public readonly Stack<GameObject> idle = new Stack<GameObject>();
        // 지금 화면에 나가 있는(활성) 오브젝트. DespawnAll 과 개수 확인에 쓴다.
        public readonly HashSet<GameObject> active = new HashSet<GameObject>();
    }

    // 풀이 만든 오브젝트 하나에 붙는 꼬리표: 어느 보관함 소속인지와 IPoolable 컴포넌트 목록.
    private struct Entry
    {
        public Bucket bucket;
        public IPoolable[] hooks;
    }

    private static PullingManager instance;

    // 씬에 배치해 둔 매니저가 있으면 그것을 쓰고, 없으면 처음 쓰는 순간 자동으로 만든다.
    // 씬에 있는 매니저의 Awake 보다 먼저 불려도 중복 생성되지 않도록 먼저 씬에서 찾아본다.
    public static PullingManager Instance
    {
        get
        {
            if (instance == null) instance = FindFirstObjectByType<PullingManager>();
            if (instance == null) instance = new GameObject("PullingManager").AddComponent<PullingManager>();
            return instance;
        }
    }

    [Tooltip("씬 시작 시 미리 만들어 둘 프리팹과 개수")]
    [SerializeField] private Preload[] preloads;

    // 프리팹 -> 보관함.
    private readonly Dictionary<GameObject, Bucket> buckets = new Dictionary<GameObject, Bucket>();
    // 풀이 만든 오브젝트 -> 꼬리표.
    private readonly Dictionary<GameObject, Entry> entries = new Dictionary<GameObject, Entry>();
    // DespawnAll 에서 순회 중 컬렉션이 바뀌는 것을 피하려고 쓰는 임시 목록.
    private readonly List<GameObject> buffer = new List<GameObject>();

    private void Awake()
    {
        // 매니저는 씬에 하나만. 뒤늦게 깨어난 중복은 스스로 사라진다.
        if (instance != null && instance != this)
        {
            Destroy(gameObject);
            return;
        }
        instance = this;

        if (preloads == null) return;
        foreach (var p in preloads)
        {
            if (p != null) Prewarm(p.prefab, p.count);
        }
    }

    // 쉬는 오브젝트가 count 개가 될 때까지 미리 만들어 둔다. 전투 중 끊김을 로딩 시점으로 옮기는 용도.
    public void Prewarm(GameObject prefab, int count)
    {
        if (prefab == null) return;
        var bucket = GetBucket(prefab);
        while (bucket.idle.Count < count) bucket.idle.Push(Create(prefab, bucket));
    }

    // 풀에서 하나 꺼내 위치/회전을 맞추고 활성화한다. 쉬는 오브젝트가 없으면 새로 만든다.
    public GameObject Spawn(GameObject prefab, Vector3 position, Quaternion rotation)
    {
        if (prefab == null) return null;
        var bucket = GetBucket(prefab);

        // 씬 전환 등으로 이미 파괴된 오브젝트가 스택에 남아 있을 수 있어 살아 있는 것이 나올 때까지 꺼낸다.
        GameObject go = null;
        while (bucket.idle.Count > 0 && go == null) go = bucket.idle.Pop();
        if (go == null) go = Create(prefab, bucket);

        go.transform.SetPositionAndRotation(position, rotation);
        bucket.active.Add(go);
        go.SetActive(true);

        var hooks = entries[go].hooks;
        for (int i = 0; i < hooks.Length; i++) hooks[i].OnSpawned();
        return go;
    }

    // 컴포넌트 타입으로 바로 받고 싶을 때 쓰는 편의 버전. 예: Spawn(enemyPrefab, pos, rot) -> Enemy
    public T Spawn<T>(T prefab, Vector3 position, Quaternion rotation) where T : Component
    {
        if (prefab == null) return null;
        var go = Spawn(prefab.gameObject, position, rotation);
        return go != null ? go.GetComponent<T>() : null;
    }

    // 오브젝트를 풀로 되돌린다.
    public void Despawn(GameObject go)
    {
        if (go == null) return;

        // 풀이 만들지 않은 오브젝트(씬에 직접 놓은 것 등)는 되돌릴 곳이 없으므로 그냥 파괴한다.
        if (!entries.TryGetValue(go, out var entry))
        {
            Destroy(go);
            return;
        }

        // 이미 되돌아간 오브젝트를 또 되돌리면 스택에 두 번 들어가므로 막는다.
        if (!entry.bucket.active.Remove(go)) return;

        for (int i = 0; i < entry.hooks.Length; i++) entry.hooks[i].OnDespawned();
        go.SetActive(false);
        entry.bucket.idle.Push(go);
    }

    // 어디서든 한 줄로 반납하는 정적 버전. 매니저가 이미 사라진 뒤(씬 종료 중)에도 새로 만들지 않고 안전하게 처리한다.
    public static void Release(GameObject go)
    {
        if (go == null) return;
        if (instance != null) instance.Despawn(go);
        else Destroy(go);
    }

    // 해당 프리팹으로 꺼낸 것을 전부 되돌린다. 화면의 탄 전체 삭제(페이즈 전환, 필살기)에 쓴다.
    public void DespawnAll(GameObject prefab)
    {
        if (prefab == null || !buckets.TryGetValue(prefab, out var bucket)) return;
        buffer.Clear();
        buffer.AddRange(bucket.active);
        for (int i = 0; i < buffer.Count; i++) Despawn(buffer[i]);
        buffer.Clear();
    }

    // 해당 프리팹으로 지금 화면에 나가 있는 개수.
    public int CountActive(GameObject prefab)
    {
        return prefab != null && buckets.TryGetValue(prefab, out var bucket) ? bucket.active.Count : 0;
    }

    private Bucket GetBucket(GameObject prefab)
    {
        if (buckets.TryGetValue(prefab, out var bucket)) return bucket;

        bucket = new Bucket { root = new GameObject(prefab.name + " Pool").transform };
        bucket.root.SetParent(transform, false);
        buckets.Add(prefab, bucket);
        return bucket;
    }

    // 새 오브젝트를 만들어 꺼둔 상태로 돌려준다. IPoolable 은 이때 한 번만 찾아 두고 계속 재사용한다.
    private GameObject Create(GameObject prefab, Bucket bucket)
    {
        var go = Instantiate(prefab, bucket.root);
        go.name = prefab.name;
        go.SetActive(false);
        entries.Add(go, new Entry { bucket = bucket, hooks = go.GetComponentsInChildren<IPoolable>(true) });
        return go;
    }

    private void OnDestroy()
    {
        if (instance == this) instance = null;
    }
}

Round Manager.cs

Assets/Scripts/Manager/Round Manager.cs

라운드(스테이지) 진행을 맡는다. 기획서 플레이루프 그대로: 일반 스테이지(잡몹 무리 여러 번) -> 보스 -> 다음 라운드 ... -> 마지막 라운드의 보스(최종 보스) 처치 시 클리어. 숫자는 전부 Inspector 에서 고칠 수 있고, 기본값은 기획서 1스테이지(찌르레기/벌매/검독수리) 값이다.

using System;
using System.Collections;
using System.Collections.Generic;
using UnityEngine;

public class RoundManager : MonoBehaviour
{
    // Ready: 시작 대기 / Wave: 잡몹 무리와 전투 중 / WaveDelay: 다음 무리를 기다리는 중
    // Boss: 보스전 / RoundClear: 라운드를 깨고 다음 라운드 대기 / GameClear: 최종 보스 처치
    public enum State { Ready, Wave, WaveDelay, Boss, RoundClear, GameClear }

    // "어떤 적을 몇 마리, 체력/점수 얼마로" 를 적는 한 줄.
    [Serializable]
    public class EnemyGroup
    {
        public EnemyKind kind = EnemyKind.Starling;
        [Tooltip("비워두면 종류별 기본 원형 적을 사용")]
        public Enemy prefab;
        public int count = 10;
        public int hp = 20;
        [Tooltip("한 마리 처치 시 점수")]
        public int score = 100;
        [Tooltip("조준탄 발사 간격(초)")]
        public float attackInterval = 0.75f;
        [Tooltip("화면 위쪽에서 얼마나 내려와 멈출지. 작을수록 뒤(위)쪽 줄")]
        public float stopFromTop = 2.5f;
    }

    // 라운드 하나의 구성.
    [Serializable]
    public class Round
    {
        public string name = "Stage 1";

        [Header("일반 스테이지")]
        [Tooltip("무리 하나에 함께 나오는 적 구성")]
        public EnemyGroup[] waveGroups;
        [Tooltip("무리가 몇 번 나오는지")]
        public int waveCount = 5;
        [Tooltip("앞 무리가 전멸한 뒤 다음 무리가 나오기까지(초)")]
        public float waveDelay = 2f;

        [Header("보스")]
        public bool hasBoss = true;
        public EnemyGroup boss;

        [Header("보스 소환")]
        [Tooltip("보스 체력이 이 비율 이하가 될 때마다 한 번씩 소환 (큰 값부터)")]
        public float[] summonHpRatios;
        public EnemyGroup[] summonGroups;
        [Tooltip("소환몹 체력 배율")]
        public float summonStatScale = 0.5f;
        [Tooltip("소환몹 점수 배율")]
        public float summonScoreScale = 1.5f;
        [Tooltip("소환 후 이 시간이 지나면 살아남은 소환몹은 퇴장(초)")]
        public float summonDuration = 10f;

        // 기획서 1스테이지 값.
        // 찌르레기: 체력 20, 공격 간격 0.75초, 10마리씩 5무리(총 50). 벌매: 체력 25, 공격 간격 1.5초, 무리마다 5마리(총 25), 찌르레기 뒤쪽.
        // 검독수리: 체력 3000. 체력 75/50/25% 에서 찌르레기 10 + 벌매 2 소환, 소환몹 능력치 50%, 10초 뒤 퇴장.
        // 점수는 기획서에 수치가 없어 흡수 점수 비율(찌르레기 1 : 벌매 2)을 따라 100 / 200, 보스 10000 으로 잡았다.
        public static Round Stage1()
        {
            return new Round
            {
                name = "Stage 1",
                waveGroups = new[]
                {
                    new EnemyGroup { kind = EnemyKind.Starling, count = 10, hp = 20, score = 100, attackInterval = 0.75f, stopFromTop = 2.6f },
                    new EnemyGroup { kind = EnemyKind.HoneyBuzzard, count = 5, hp = 25, score = 200, attackInterval = 1.5f, stopFromTop = 1.3f },
                },
                waveCount = 5,
                waveDelay = 2f,
                hasBoss = true,
                boss = new EnemyGroup { kind = EnemyKind.GoldenEagle, count = 1, hp = 3000, score = 10000, attackInterval = 0.75f, stopFromTop = 2f },
                summonHpRatios = new[] { 0.75f, 0.5f, 0.25f },
                summonGroups = new[]
                {
                    new EnemyGroup { kind = EnemyKind.Starling, count = 10, hp = 20, score = 100, attackInterval = 0.75f, stopFromTop = 3.8f },
                    new EnemyGroup { kind = EnemyKind.HoneyBuzzard, count = 2, hp = 25, score = 200, attackInterval = 1.5f, stopFromTop = 3.2f },
                },
                summonStatScale = 0.5f,
                summonScoreScale = 1.5f,
                summonDuration = 10f,
            };
        }
    }

    public static RoundManager Instance { get; private set; }

    [Tooltip("씬 시작 후 첫 무리가 나오기까지(초)")]
    [SerializeField] private float firstDelay = 1.5f;
    [Tooltip("보스를 잡은 뒤 다음 라운드가 시작되기까지(초)")]
    [SerializeField] private float roundDelay = 3f;
    [Tooltip("라운드 목록. 마지막 라운드의 보스가 최종 보스")]
    [SerializeField] private Round[] rounds = { Round.Stage1() };

    [Header("기본 적 모습 (그룹의 Prefab 칸을 비웠을 때 사용)")]
    // Assets/Sprites/Enemies 의 날갯짓 2프레임씩. 아래(플레이어 쪽)를 바라보게 그려져 있다.
    [SerializeField] private Sprite[] starlingFrames;
    [SerializeField] private Sprite[] honeyBuzzardFrames;
    [SerializeField] private Sprite[] goldenEagleFrames;

    // UI 나 연출이 구독하는 알림들. 라운드/무리 번호는 1부터 센다.
    public event Action<int> OnRoundStarted;
    // (현재 무리 번호, 전체 무리 수)
    public event Action<int, int> OnWaveStarted;
    public event Action<Enemy> OnBossSpawned;
    public event Action<int> OnRoundCleared;
    public event Action OnGameCleared;

    public State CurrentState { get; private set; } = State.Ready;
    public int CurrentRound { get; private set; }
    public int RoundCount => rounds != null ? rounds.Length : 0;
    public int CurrentWave { get; private set; }
    public int WaveCount { get; private set; }
    public Enemy Boss { get; private set; }
    // 지금 살아 있는 일반 스테이지 잡몹 수.
    public int AliveCount => alive.Count;

    // 일반 스테이지 무리에서 살아 있는 적. 비면 다음 무리로 넘어간다.
    private readonly List<Enemy> alive = new List<Enemy>();
    // 보스가 소환해서 살아 있는 적. 보스가 죽으면 한꺼번에 퇴장시킨다.
    private readonly List<Enemy> summoned = new List<Enemy>();
    // 프리팹을 비워 둔 종류를 위한 기본 적 틀(종류별 1개).
    private readonly Dictionary<EnemyKind, Enemy> templates = new Dictionary<EnemyKind, Enemy>();
    // 목록을 순회하면서 원본이 바뀌는 것을 피하려는 임시 목록.
    private readonly List<Enemy> buffer = new List<Enemy>();

    private void Awake()
    {
        Instance = this;
    }

    private void OnDestroy()
    {
        if (Instance == this) Instance = null;
    }

    // 전체 진행. 코루틴 하나가 위에서 아래로 읽히는 순서 그대로 게임이 흘러간다.
    private IEnumerator Start()
    {
        ScoreManager.Instance.ResetScore();
        yield return new WaitForSeconds(firstDelay);

        for (int r = 0; r < rounds.Length; r++)
        {
            var round = rounds[r];
            CurrentRound = r + 1;
            OnRoundStarted?.Invoke(CurrentRound);

            yield return RunWaves(round);
            if (round.hasBoss && round.boss != null) yield return RunBoss(round);

            OnRoundCleared?.Invoke(CurrentRound);

            if (r < rounds.Length - 1)
            {
                CurrentState = State.RoundClear;
                yield return new WaitForSeconds(roundDelay);
            }
        }

        CurrentState = State.GameClear;
        OnGameCleared?.Invoke();
    }

    // 일반 스테이지: 무리를 내보내고, 전멸하면 waveDelay 뒤에 다음 무리를 내보낸다.
    private IEnumerator RunWaves(Round round)
    {
        WaveCount = round.waveGroups != null && round.waveGroups.Length > 0 ? round.waveCount : 0;

        for (int w = 1; w <= WaveCount; w++)
        {
            CurrentState = State.Wave;
            CurrentWave = w;
            OnWaveStarted?.Invoke(w, WaveCount);

            foreach (var group in round.waveGroups) SpawnGroup(group, 1f, 1f, false, alive);

            while (alive.Count > 0) yield return null;

            if (w < WaveCount)
            {
                CurrentState = State.WaveDelay;
                yield return new WaitForSeconds(round.waveDelay);
            }
        }
    }

    // 보스전: 잡몹이 다 죽으면 바로 등장한다. 싸우는 동안 체력 비율을 지켜보다 기준을 넘으면 잡몹을 소환한다.
    private IEnumerator RunBoss(Round round)
    {
        CurrentState = State.Boss;

        bool bossAlive = true;
        Boss = SpawnOne(round.boss, 0.5f, 1f, 1f, false);
        Boss.OnRemoved += _ => bossAlive = false;
        OnBossSpawned?.Invoke(Boss);

        int nextSummon = 0;
        while (bossAlive)
        {
            // 한 번에 큰 피해를 받아 기준 두 개를 동시에 넘으면 소환도 두 번 일어나야 하므로 while 로 확인한다.
            while (round.summonHpRatios != null && nextSummon < round.summonHpRatios.Length
                   && Boss.HpRatio <= round.summonHpRatios[nextSummon])
            {
                nextSummon++;
                StartCoroutine(Summon(round));
            }
            yield return null;
        }

        Boss = null;

        // 보스가 죽으면 남은 소환몹은 싸울 이유가 없으므로 퇴장시킨다.
        buffer.Clear();
        buffer.AddRange(summoned);
        foreach (var e in buffer) e.Leave();
        buffer.Clear();
    }

    // 소환 한 번: 약해진 잡몹을 부르고, summonDuration 뒤에 그 회차에서 살아남은 것만 퇴장시킨다.
    private IEnumerator Summon(Round round)
    {
        if (round.summonGroups == null) yield break;

        var batch = new List<Enemy>();
        foreach (var group in round.summonGroups)
            SpawnGroup(group, round.summonStatScale, round.summonScoreScale, true, batch);

        yield return new WaitForSeconds(round.summonDuration);

        foreach (var e in batch.ToArray()) e.Leave();
    }

    // 한 그룹을 가로로 고르게 벌려 내보내고, tracker 목록에 넣어 살아 있는 수를 센다.
    private void SpawnGroup(EnemyGroup group, float hpScale, float scoreScale, bool isSummon, List<Enemy> tracker)
    {
        if (group == null) return;

        for (int i = 0; i < group.count; i++)
        {
            float t = (i + 1f) / (group.count + 1f);
            var e = SpawnOne(group, t, hpScale, scoreScale, isSummon);

            tracker.Add(e);
            e.OnRemoved += dead => tracker.Remove(dead);

            if (isSummon)
            {
                summoned.Add(e);
                e.OnRemoved += dead => summoned.Remove(dead);
            }
        }
    }

    // 적 한 마리를 풀에서 꺼낸다. t 는 가로 위치(0 = 왼쪽 끝, 1 = 오른쪽 끝).
    private Enemy SpawnOne(EnemyGroup group, float t, float hpScale, float scoreScale, bool isSummon)
    {
        var cam = Camera.main;
        float h = cam != null ? cam.orthographicSize : 5f;
        float w = cam != null ? h * cam.aspect : 9f;
        Vector2 c = cam != null ? (Vector2)cam.transform.position : Vector2.zero;

        // 화면 폭의 80% 안쪽에 배치하고, 화면 밖 위에서 target 까지 내려오게 한다.
        float x = c.x + Mathf.Lerp(-w, w, t) * 0.8f;
        Vector2 target = new Vector2(x, c.y + h - group.stopFromTop);
        Vector2 start = new Vector2(x, c.y + h + 1f);

        var e = PullingManager.Instance.Spawn(TemplateFor(group), start, Quaternion.identity);
        e.Configure(group.kind, Mathf.RoundToInt(group.score * scoreScale), isSummon, group.attackInterval);
        // 프리팹을 넣었으면 프리팹에 설정한 패턴을, 비웠으면 종류별 기본 패턴을 쓴다.
        e.Setup(target, Mathf.RoundToInt(group.hp * hpScale), group.prefab != null ? null : DefaultPatterns(group.kind));
        return e;
    }

    // 프리팹이 비어 있을 때의 패턴. 보스는 기존 4가지 패턴을 돌리고, 잡몹은 조준탄만 쏜다.
    private static Enemy.Pattern[] DefaultPatterns(EnemyKind kind)
    {
        if (kind == EnemyKind.GoldenEagle)
            return new[] { Enemy.Pattern.Ring, Enemy.Pattern.AimedFan, Enemy.Pattern.Spiral, Enemy.Pattern.Curve };
        return new[] { Enemy.Pattern.AimedShot };
    }

    // 프리팹이 있으면 그것을, 없으면 종류별 색/크기의 원형 적을 꺼진 상태로 하나 만들어 프리팹처럼 쓴다.
    private Enemy TemplateFor(EnemyGroup group)
    {
        if (group.prefab != null) return group.prefab;
        if (templates.TryGetValue(group.kind, out var template)) return template;

        // 종류별 값: 날갯짓 프레임, 스프라이트를 쓸 때의 크기/판정 반지름, 스프라이트가 없을 때 쓰는 원의 크기/색.
        Sprite[] frames;
        float spriteScale, spriteHit, circleSize;
        Color circleColor;
        switch (group.kind)
        {
            case EnemyKind.HoneyBuzzard:
                frames = honeyBuzzardFrames; spriteScale = 1f; spriteHit = 0.4f;
                circleSize = 0.7f; circleColor = new Color(0.95f, 0.75f, 0.2f);
                break;
            case EnemyKind.GoldenEagle:
                frames = goldenEagleFrames; spriteScale = 1.5f; spriteHit = 1f;
                circleSize = 2.2f; circleColor = new Color(0.55f, 0.35f, 0.15f);
                break;
            default:
                frames = starlingFrames; spriteScale = 1f; spriteHit = 0.25f;
                circleSize = 0.45f; circleColor = new Color(0.9f, 0.2f, 0.5f);
                break;
        }
        bool hasSprite = frames != null && frames.Length > 0 && frames[0] != null;

        var go = new GameObject(group.kind.ToString());
        go.SetActive(false);
        go.transform.SetParent(transform, false);
        go.transform.localScale = Vector3.one * (hasSprite ? spriteScale : circleSize);

        // 스프라이트는 이미 색이 칠해져 있으므로 흰색(원본 그대로)으로 그린다.
        var sr = go.AddComponent<SpriteRenderer>();
        sr.sprite = hasSprite ? frames[0] : EnemyBulletPool.CircleSprite;
        sr.color = hasSprite ? Color.white : circleColor;
        sr.sortingOrder = 10;

        template = go.AddComponent<Enemy>();
        // 원 스프라이트는 지름이 1 유닛이라 판정 반지름이 크기의 절반이고, 새 스프라이트는 몸통 기준으로 따로 잡았다.
        template.HitRadius = hasSprite ? spriteHit : circleSize * 0.5f;
        if (hasSprite) template.SetFrames(frames, group.kind == EnemyKind.GoldenEagle ? 4f : 7f);
        templates.Add(group.kind, template);
        return template;
    }
}

using System;
using UnityEngine;

public class ScoreManager : MonoBehaviour
{
    private static ScoreManager instance;

    // 씬에 배치하지 않아도 처음 쓰는 순간 만들어진다. 그래서 적이 죽을 때 null 검사 없이 점수를 줄 수 있다.
    public static ScoreManager Instance
    {
        get
        {
            if (instance == null) instance = FindFirstObjectByType<ScoreManager>();
            if (instance == null) instance = new GameObject("ScoreManager").AddComponent<ScoreManager>();
            return instance;
        }
    }

    // Inspector 에서 현재 값을 눈으로 확인할 수 있도록 직렬화한다.
    [field: SerializeField] public int Score { get; private set; }
    [field: SerializeField] public int HighScore { get; private set; }
    // 이번 판에 처치한 적 수.
    [field: SerializeField] public int KillCount { get; private set; }

    // 점수가 바뀔 때 (현재 점수, 최고 점수)를 알려준다. 점수 UI 가 구독한다.
    public event Action<int, int> OnScoreChanged;

    private const string HighScoreKey = "HighScore";

    private void Awake()
    {
        if (instance != null && instance != this)
        {
            Destroy(gameObject);
            return;
        }
        instance = this;

        // 씬이 바뀌어도(게임 -> 랭킹) 점수가 남아 있도록 유지한다.
        DontDestroyOnLoad(gameObject);
        HighScore = PlayerPrefs.GetInt(HighScoreKey, 0);
    }

    // 적 처치 점수. 처치 수를 함께 센 뒤 AddScore 로 넘긴다.
    public void AddKillScore(int amount)
    {
        KillCount++;
        AddScore(amount);
    }

    public void AddScore(int amount)
    {
        // 0 이하 값은 무시한다. 감점이 필요해지면 이 줄을 지운다.
        if (amount <= 0) return;

        Score += amount;

        if (Score > HighScore)
        {
            HighScore = Score;
            PlayerPrefs.SetInt(HighScoreKey, HighScore);
            PlayerPrefs.Save();
        }

        OnScoreChanged?.Invoke(Score, HighScore);
    }

    // 새 판을 시작할 때 현재 점수와 처치 수를 0으로 되돌린다. 최고 점수는 유지된다.
    public void ResetScore()
    {
        Score = 0;
        KillCount = 0;
        OnScoreChanged?.Invoke(Score, HighScore);
    }

    private void OnDestroy()
    {
        if (instance == this) instance = null;
    }
}

using System.Collections;
using System.Collections.Generic;
using UnityEngine;

// 기획서에 나오는 적 종류. 찌르레기(무리형), 벌매(분열 탄막형), 검독수리(보스).
public enum EnemyKind { Starling, HoneyBuzzard, GoldenEagle }

public class Enemy : MonoBehaviour, IPoolable
{
    // AimedShot 은 기획서 찌르레기의 "플레이어를 조준한 직선 탄 1발"이다. 나머지는 기존 패턴.
    public enum Pattern { Ring, AimedFan, Spiral, Curve, AimedShot }

    // 지금 화면에 살아 있는 적 목록. 유도탄이 타겟을 찾을 때 쓴다.
    public static readonly List<Enemy> Active = new List<Enemy>();

    // 에디터에서 플레이를 다시 시작할 때 이전 판의 목록이 남지 않게 비운다.
    [RuntimeInitializeOnLoadMethod(RuntimeInitializeLoadType.SubsystemRegistration)]
    private static void ResetStatics()
    {
        Active.Clear();
    }

    [Header("종류 / 점수")]
    [SerializeField] private EnemyKind kind = EnemyKind.Starling;
    [Tooltip("처치 시 플레이어가 얻는 점수. RoundManager 가 스폰할 때 덮어씀")]
    [SerializeField] private int scoreValue = 100;

    [Header("체력 / 피격")]
    [SerializeField] private int maxHp = 60;
    [Tooltip("플레이어 탄에 맞는 범위")]
    [SerializeField] private float hitRadius = 0.4f;
    [Tooltip("플레이어 탄 한 발에 깎이는 체력. 캐릭터별 공격력(3~5)이 PlayerData 에 생기면 그 값으로 교체")]
    [SerializeField] private int damagePerHit = 4;

    [Header("이동")]
    [SerializeField] private float enterSpeed = 3f;
    [SerializeField] private float swayWidth = 1f;
    [SerializeField] private float swaySpeed = 0.6f;

    [Header("패턴")]
    [SerializeField] private Pattern[] patterns = { Pattern.Ring, Pattern.AimedFan, Pattern.Spiral, Pattern.Curve };
    [SerializeField] private float patternDuration = 4f;
    [SerializeField] private float patternRest = 0.8f;
    [SerializeField] private float bulletSpeed = 2.5f;
    [Tooltip("AimedShot 패턴의 발사 간격(초). 기획서: 찌르레기 0.75, 벌매 1.5")]
    [SerializeField] private float aimedShotInterval = 0.75f;

    [Header("모습")]
    [Tooltip("날갯짓 프레임. 2장 이상이면 순서대로 반복 재생")]
    [SerializeField] private Sprite[] frames;
    [Tooltip("초당 프레임 수")]
    [SerializeField] private float frameRate = 6f;

    // 플레이어에게 처치됐을 때만 불린다. 점수/보상처럼 "잡았을 때"만 일어나야 하는 일에 쓴다.
    public event System.Action<Enemy> OnDead;
    // 처치든 퇴장이든 화면에서 사라질 때 항상 불린다. "남은 적 수" 계산에 쓴다.
    public event System.Action<Enemy> OnRemoved;

    public EnemyKind Kind => kind;
    public int ScoreValue => scoreValue;
    // 보스가 소환한 잡몹인지. 소환몹은 탄이 파란색이고 일정 시간 뒤 퇴장한다.
    public bool IsSummoned { get; private set; }
    public int Hp => hp;
    public int MaxHp => maxHp;
    // 남은 체력 비율(0~1). 보스의 75/50/25% 소환 판정에 쓴다.
    public float HpRatio => maxHp > 0 ? (float)hp / maxHp : 0f;
    public float HitRadius { get => hitRadius; set => hitRadius = value; }

    // 스포너가 기본 적 틀을 만들 때 날갯짓 프레임을 넣어 준다. 직렬화 필드라서 풀에서 복제될 때 함께 복사된다.
    public void SetFrames(Sprite[] frames, float frameRate)
    {
        this.frames = frames;
        this.frameRate = frameRate;
    }

    private int hp;
    private Vector2 anchor;
    // 자리에 도착했는지. 도착한 뒤에만 좌우로 흔들린다.
    private bool arrived;
    // 행동 코루틴을 이미 시작했는지. Setup 과 Start 가 둘 다 시작하지 않게 막는다.
    private bool begun;
    // 퇴장 중인지.
    private bool leaving;
    // 이미 사라지는 처리를 했는지. 같은 프레임에 두 번 죽는 것을 막는다.
    private bool removed;
    private SpriteRenderer sr;
    private Color baseColor;
    private float flashTimer;
    // 무리 전체가 똑같은 박자로 날갯짓하지 않도록 개체마다 다른 시작 시점.
    private float frameOffset;

    private void Awake()
    {
        hp = maxHp;
        anchor = transform.position;
        sr = GetComponent<SpriteRenderer>();
        if (sr != null) baseColor = sr.color;
    }

    private void OnEnable() => Active.Add(this);
    private void OnDisable() => Active.Remove(this);

    // 풀에서 꺼내질 때: 이전 생의 상태를 전부 지운다.
    public void OnSpawned()
    {
        arrived = false;
        begun = false;
        leaving = false;
        removed = false;
        flashTimer = 0f;
        IsSummoned = false;
        frameOffset = Random.value;
        if (sr != null) sr.color = baseColor;
    }

    // 풀로 돌아갈 때: 이전 스포너가 걸어 둔 구독이 다음 생에 따라오지 않게 끊는다.
    public void OnDespawned()
    {
        StopAllCoroutines();
        OnDead = null;
        OnRemoved = null;
    }

    // 스포너가 Setup 보다 먼저 부른다. 종류, 처치 점수, 소환몹 여부, 조준탄 발사 간격을 정한다.
    public void Configure(EnemyKind kind, int scoreValue, bool summoned, float aimedShotInterval)
    {
        this.kind = kind;
        this.scoreValue = scoreValue;
        IsSummoned = summoned;
        if (aimedShotInterval > 0f) this.aimedShotInterval = aimedShotInterval;
    }

    // 스포너가 꺼낸 직후 부른다. target 까지 내려온 뒤 패턴을 시작한다. patterns 가 null 이면 프리팹에 설정된 패턴을 쓴다.
    public void Setup(Vector2 target, int hp, Pattern[] patterns = null)
    {
        anchor = target;
        maxHp = this.hp = Mathf.Max(1, hp);
        if (patterns != null && patterns.Length > 0) this.patterns = patterns;
        Begin();
    }

    // 스포너 없이 씬에 직접 놓은 적도 움직이도록, Setup 이 불리지 않았으면 여기서 시작한다.
    private void Start()
    {
        if (!begun) Begin();
    }

    private void Begin()
    {
        begun = true;
        StopAllCoroutines();
        StartCoroutine(Run());
    }

    private IEnumerator Run()
    {
        while (((Vector2)transform.position - anchor).sqrMagnitude > 0.0001f)
        {
            transform.position = Vector2.MoveTowards(transform.position, anchor, enterSpeed * Time.deltaTime);
            yield return null;
        }
        arrived = true;

        if (patterns == null || patterns.Length == 0) yield break;

        // 같은 무리가 한 박자에 동시에 쏘지 않도록 첫 발사를 조금씩 어긋나게 한다.
        yield return new WaitForSeconds(Random.value * aimedShotInterval);

        int i = 0;
        while (true)
        {
            yield return RunPattern(patterns[i % patterns.Length]);
            yield return new WaitForSeconds(patternRest);
            i++;
        }
    }

    private void Update()
    {
        if (arrived && !leaving)
        {
            float x = Mathf.Sin(Time.time * swaySpeed * Mathf.PI) * swayWidth;
            transform.position = anchor + new Vector2(x, 0f);
        }

        CheckPlayerBullets();

        // 날갯짓: 시간에 따라 프레임을 번갈아 보여 준다.
        if (sr != null && frames != null && frames.Length > 1)
        {
            int index = (int)((Time.time + frameOffset) * frameRate) % frames.Length;
            if (frames[index] != null) sr.sprite = frames[index];
        }

        if (flashTimer > 0f)
        {
            flashTimer -= Time.deltaTime;
            if (flashTimer <= 0f && sr != null) sr.color = baseColor;
        }
    }

    // 플레이어 탄 목록을 돌며 원 대 점 거리로 피격을 검사한다.
    private void CheckPlayerBullets()
    {
        var list = PlayerBullets.Active;
        Vector2 pos = transform.position;
        float r2 = hitRadius * hitRadius;

        for (int i = list.Count - 1; i >= 0; i--)
        {
            var b = list[i];
            if (b == null || ((Vector2)b.position - pos).sqrMagnitude > r2) continue;

            // 맞은 탄은 바로 풀로 돌려보낸다. 꺼지면서 목록에서도 빠지므로 같은 프레임에 다른 적이 또 맞지 않는다.
            PullingManager.Release(b.gameObject);
            TakeDamage(damagePerHit);
            if (removed) return;
        }
    }

    public void TakeDamage(int damage)
    {
        if (removed || hp <= 0) return;
        hp -= damage;

        if (sr != null)
        {
            // 피격 번쩍임. 스프라이트를 쓰는 적은 원래 색이 흰색이라 흰색으로는 티가 안 나므로 붉게 물들인다.
            sr.color = baseColor == Color.white ? new Color(1f, 0.45f, 0.45f) : Color.white;
            flashTimer = 0.05f;
        }

        if (hp <= 0) Remove(true);
    }

    // 처치되지 않은 채 화면 위로 퇴장한다. 기획서: 소환 후 10초가 지나면 살아남은 잡몹은 퇴장하며 점수를 주지 않는다.
    public void Leave()
    {
        if (removed || leaving) return;
        leaving = true;
        StopAllCoroutines();
        StartCoroutine(LeaveRoutine());
    }

    private IEnumerator LeaveRoutine()
    {
        var cam = Camera.main;
        float top = cam != null ? cam.transform.position.y + cam.orthographicSize + 1f : transform.position.y + 12f;

        while (transform.position.y < top)
        {
            transform.position += Vector3.up * (enterSpeed * 2f * Time.deltaTime);
            yield return null;
        }
        Remove(false);
    }

    // 사라지는 처리를 한 곳으로 모았다. killed 가 true 일 때만 점수를 주고 OnDead 를 알린다.
    private void Remove(bool killed)
    {
        if (removed) return;
        removed = true;

        if (killed)
        {
            ScoreManager.Instance.AddKillScore(scoreValue);
            OnDead?.Invoke(this);
        }
        OnRemoved?.Invoke(this);

        PullingManager.Release(gameObject);
    }

    private IEnumerator RunPattern(Pattern p)
    {
        switch (p)
        {
            case Pattern.Ring: return Ring();
            case Pattern.AimedFan: return AimedFan();
            case Pattern.Spiral: return Spiral();
            case Pattern.AimedShot: return AimedShot();
            default: return Curve();
        }
    }

    // 전방위 원형탄. 발사할 때마다 반 칸씩 엇갈려서 빠져나갈 틈이 생긴다.
    private IEnumerator Ring()
    {
        const int count = 24;
        float step = 360f / count;
        float offset = 0f;
        float end = Time.time + patternDuration;
        while (Time.time < end)
        {
            for (int i = 0; i < count; i++)
                Fire(offset + step * i, bulletSpeed, new Color(1f, 0.35f, 0.35f), BulletShape.Orb);
            offset += step * 0.5f;
            yield return new WaitForSeconds(0.45f);
        }
    }

    // 플레이어를 조준한 5갈래 부채꼴을 3연사한다.
    private IEnumerator AimedFan()
    {
        const int ways = 5;
        const float spread = 14f;
        float end = Time.time + patternDuration;
        while (Time.time < end)
        {
            float aim = AimAngle();
            for (int burst = 0; burst < 3; burst++)
            {
                for (int i = 0; i < ways; i++)
                    Fire(aim + (i - (ways - 1) * 0.5f) * spread, bulletSpeed * 1.6f, new Color(0.4f, 0.7f, 1f), BulletShape.Feather);
                yield return new WaitForSeconds(0.08f);
            }
            yield return new WaitForSeconds(0.6f);
        }
    }

    // 3갈래로 돌아가는 나선.
    private IEnumerator Spiral()
    {
        const int arms = 3;
        float angle = 0f;
        float end = Time.time + patternDuration;
        while (Time.time < end)
        {
            for (int i = 0; i < arms; i++)
                Fire(angle + 360f / arms * i, bulletSpeed, new Color(1f, 0.85f, 0.3f), BulletShape.Rice);
            angle += 11f;
            yield return new WaitForSeconds(0.05f);
        }
    }

    // 좌우로 휘어지는 탄. 느리게 출발해 가속하므로 타이밍이 어긋난다.
    private IEnumerator Curve()
    {
        const int count = 12;
        float step = 360f / count;
        bool flip = false;
        float end = Time.time + patternDuration;
        while (Time.time < end)
        {
            float turn = flip ? -35f : 35f;
            for (int i = 0; i < count; i++)
                Fire(step * i, bulletSpeed * 0.6f, new Color(0.75f, 0.45f, 1f), BulletShape.Star, 0.5f, turn);
            flip = !flip;
            yield return new WaitForSeconds(0.35f);
        }
    }

    // 발사 순간의 플레이어 위치를 조준해 한 발씩 쏜다. 기획서 찌르레기 기본 공격.
    private IEnumerator AimedShot()
    {
        float end = Time.time + patternDuration;
        while (Time.time < end)
        {
            Fire(AimAngle(), bulletSpeed, new Color(1f, 0.6f, 0.3f), BulletShape.Rice);
            yield return new WaitForSeconds(aimedShotInterval);
        }
    }

    // 기획서: 소환된 잡몹의 탄은 파란색으로 해서 보스 탄과 구분한다.
    private void Fire(float angle, float speed, Color color, BulletShape shape, float accel = 0f, float angularSpeed = 0f)
    {
        if (IsSummoned) color = new Color(0.3f, 0.55f, 1f);
        EnemyBulletPool.Instance.Spawn(transform.position, angle, speed, color, accel, angularSpeed, shape);
    }

    // 플레이어 판정점을 향한 각도. 플레이어가 없으면 아래(-90도)로 쏜다.
    private float AimAngle()
    {
        var player = PlayerHit.Instance;
        if (player == null) return -90f;
        Vector2 d = player.HitPosition - (Vector2)transform.position;
        return Mathf.Atan2(d.y, d.x) * Mathf.Rad2Deg;
    }

    private void OnDrawGizmosSelected()
    {
        Gizmos.color = Color.yellow;
        Gizmos.DrawWireSphere(transform.position, hitRadius);
    }
}

        float w = h * cam.aspect;

        Bounds = new Rect(c.x - w - boundsMargin, c.y - h - boundsMargin, (w + boundsMargin) * 2f, (h + boundsMargin) * 2f);

        go.transform.localScale = Vector3.one * (orbSprite != null ? spriteBulletSize : defaultBulletSize);

        var sr = go.AddComponent<SpriteRenderer>();
        sr.sprite = orbSprite != null ? orbSprite : CircleSprite;
        sr.sortingOrder = 20;

        return go.AddComponent<EnemyBullet>();

    // 지름 1 유닛짜리 원 스프라이트. 한 번만 그려서 계속 재사용한다.
    public static Sprite CircleSprite
    {
        get
        {
            if (circleSprite != null) return circleSprite;

            const int size = 64;
            var tex = new Texture2D(size, size, TextureFormat.RGBA32, false) { filterMode = FilterMode.Bilinear };
            float r = size * 0.5f;
            for (int y = 0; y < size; y++)
            {
                for (int x = 0; x < size; x++)
                {
                    // d: 중심에서의 거리(0 = 중심, 1 = 가장자리). 가장자리는 부드럽게, 안쪽 심지는 더 밝게.
                    float d = Vector2.Distance(new Vector2(x + 0.5f, y + 0.5f), new Vector2(r, r)) / r;
                    float a = Mathf.Clamp01((1f - d) * 12f);
                    float core = d < 0.55f ? 1f : 0.75f;
                    tex.SetPixel(x, y, new Color(core, core, core, a));
                }
            }
            tex.Apply();
            circleSprite = Sprite.Create(tex, new Rect(0, 0, size, size), new Vector2(0.5f, 0.5f), size);
            return circleSprite;
        }
    }

    private void OnDestroy()
    {
        if (instance == this) instance = null;
    }

using UnityEngine;

// 각도 기준: 0 = 오른쪽, 90 = 위, -90 = 아래
public class EnemyBullet : MonoBehaviour
{
    [Tooltip("판정 반지름. 그림보다 작게 잡아야 억울한 피격이 없음")]
    [SerializeField] private float radius = 0.08f;
    [SerializeField] private int damage = 1;
    [SerializeField] private float maxLifeTime = 20f;

    private SpriteRenderer sr;
    private float angle;
    private float speed;
    private float accel;
    private float angularSpeed;
    private float life;

    public float Radius => radius;

    private void Awake()
    {
        sr = GetComponent<SpriteRenderer>();
    }

    // 풀에서 꺼낼 때 탄 모양(구슬, 쌀알, 깃털, 별)을 바꿔 끼운다. null 이면 지금 모양을 유지한다.
    public void SetSprite(Sprite sprite)
    {
        if (sr != null && sprite != null) sr.sprite = sprite;
    }

    public void Init(Vector2 pos, float angle, float speed, Color color, float accel, float angularSpeed)
    {
        transform.position = pos;
        this.angle = angle;
        this.speed = speed;
        this.accel = accel;
        this.angularSpeed = angularSpeed;
        life = 0f;
        if (sr != null) sr.color = color;
        ApplyRotation();
    }

    private void Update()
    {
        float dt = Time.deltaTime;
        life += dt;
        speed = Mathf.Max(0f, speed + accel * dt);
        angle += angularSpeed * dt;

        float rad = angle * Mathf.Deg2Rad;
        transform.position += new Vector3(Mathf.Cos(rad), Mathf.Sin(rad), 0f) * (speed * dt);
        if (angularSpeed != 0f) ApplyRotation();

        if (HitPlayer()) return;

        if (life > maxLifeTime || !EnemyBulletPool.Bounds.Contains(transform.position))
            EnemyBulletPool.Instance.Despawn(this);
    }

    // 원 대 원 거리 판정. 플레이어 쪽 판정 크기는 PlayerHit.HitRadius
    private bool HitPlayer()
    {
        var player = PlayerHit.Instance;
        if (player == null || !player.CanBeHit) return false;

        float r = radius + player.HitRadius;
        if (((Vector2)transform.position - player.HitPosition).sqrMagnitude > r * r) return false;

        player.TakeHit(damage);
        EnemyBulletPool.Instance.Despawn(this);
        return true;
    }

    // 탄 스프라이트는 위쪽을 바라보게 그려야 진행 방향과 맞음
    private void ApplyRotation()
    {
        transform.rotation = Quaternion.Euler(0f, 0f, angle - 90f);
    }

    private void OnDrawGizmosSelected()
    {
        Gizmos.color = Color.red;
        Gizmos.DrawWireSphere(transform.position, radius);
    }
}

using UnityEngine;

public class HomingBullet : MonoBehaviour, IPoolable
{
    [SerializeField] private float speed = 5f;
    // 풀로 반납되기까지의 시간(초).
    [SerializeField] private float destroyTime = 5f;
    [SerializeField] private Transform target;

    private float life;

    // 재사용될 때 이전에 쫓던 적이 남아 있지 않도록 타겟과 수명을 비운다.
    public void OnSpawned()
    {
        life = 0f;
        target = null;
    }

    public void OnDespawned() => target = null;

    // 적이 피격 검사를 할 수 있도록 활성 탄 목록에 등록/해제한다.
    private void OnEnable() => PlayerBullets.Active.Add(transform);
    private void OnDisable() => PlayerBullets.Active.Remove(transform);

    private void Update()
    {
        // 적은 죽어도 파괴되지 않고 풀로 돌아가 꺼지기만 하므로 null 검사만으로는 부족하다. 꺼진 타겟도 놓아준다.
        if (target == null || !target.gameObject.activeInHierarchy) FindNearestEnemy();

        if (target != null)
        {
            Vector2 direction = ((Vector2)target.position - (Vector2)transform.position).normalized;
            transform.position += (Vector3)(direction * (speed * Time.deltaTime));
        }
        else
        {
            transform.Translate(speed * Time.deltaTime * Vector3.up);
        }

        life += Time.deltaTime;
        if (life >= destroyTime) PullingManager.Release(gameObject);
    }

    // 태그 검색 대신 Enemy.Active 목록을 쓴다. 매 프레임 불려도 씬 전체를 뒤지지 않는다.
    private void FindNearestEnemy()
    {
        target = null;
        float shortest = Mathf.Infinity;
        Vector2 pos = transform.position;

        var enemies = Enemy.Active;
        for (int i = 0; i < enemies.Count; i++)
        {
            float d = ((Vector2)enemies[i].transform.position - pos).sqrMagnitude;
            if (d >= shortest) continue;
            shortest = d;
            target = enemies[i].transform;
        }
    }
}

using UnityEngine;
using UnityEngine.InputSystem;

public class PlayerShooting : MonoBehaviour
{
    [SerializeField] private GameObject bulletPrefab;
    [SerializeField] private GameObject homingbulletPrefab;
    [SerializeField] private Transform firePoint;

    [Header("발사 설정")]
    // 공격 키를 누르고 있을 때의 연사 간격(초).
    [SerializeField] private float fireRate = 0.15f;
    private float nextFireTime;

    private void Update()
    {
        var kb = Keyboard.current;
        if (kb == null) return;

        // KeyManager 에 설정된 키를 쓰되, 없으면 기본 키(Z, X)를 쓴다.
        Key attackKey = Key.Z;
        Key ultimateKey = Key.X;

        if (KeyManager.Instance != null && KeyManager.Instance.keys != null)
        {
            if (KeyManager.Instance.keys.TryGetValue(KeyAction.Attack, out var atk)) attackKey = atk;
            if (KeyManager.Instance.keys.TryGetValue(KeyAction.Ultimate, out var ult)) ultimateKey = ult;
        }

        // 기본 공격: 누르고 있으면 fireRate 간격으로 연사.
        if (kb[attackKey].isPressed && Time.time >= nextFireTime)
        {
            nextFireTime = Time.time + fireRate;
            Shoot();
        }

        // 유도탄: 누를 때마다 한 발.
        if (kb[ultimateKey].wasPressedThisFrame)
        {
            HomingShoot();
        }
    }

    private void Shoot()
    {
        if (bulletPrefab == null)
        {
            Debug.LogWarning("[PlayerShooting] Bullet Prefab이 Inspector에 할당되지 않았습니다!");
            return;
        }

        Vector3 spawnPos = firePoint != null ? firePoint.position : transform.position;
        PullingManager.Instance.Spawn(bulletPrefab, spawnPos, Quaternion.identity);
    }

    private void HomingShoot()
    {
        // 유도탄 프리팹이 비어 있으면 일반 탄으로 대신 쏴서 동작은 확인할 수 있게 한다.
        GameObject prefabToSpawn = homingbulletPrefab != null ? homingbulletPrefab : bulletPrefab;
        if (prefabToSpawn == null)
        {
            Debug.LogWarning("[PlayerShooting] 발사할 총알 Prefab이 Inspector에 할당되지 않았습니다!");
            return;
        }

        if (homingbulletPrefab == null)
        {
            Debug.LogWarning("[PlayerShooting] Homingbullet Prefab이 할당되지 않아 일반 Bullet으로 대체 발사했습니다. Inspector에서 Homingbullet Prefab을 할당해주세요.");
        }

        Vector3 spawnPos = firePoint != null ? firePoint.position : transform.position;
        PullingManager.Instance.Spawn(prefabToSpawn, spawnPos, Quaternion.identity);
    }
}
